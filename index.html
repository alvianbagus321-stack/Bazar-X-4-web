} const targetEmail = String( sheet .getRange(targetRow, 2) .getValue() || "" ) .toLowerCase(); const requesterEmail = String( profile.email || "" ) .toLowerCase(); /* * Boleh menghapus jika: * * 1. Admin * atau * 2. Pemilik komentar */ const allowed = requesterEmail === ADMIN_EMAIL.toLowerCase() || requesterEmail === targetEmail; if (!allowed) { return postResult_({ ok: false, error: "Kamu tidak punya izin menghapus ulasan ini." }); } /* * Hapus foto dari Drive */ const photoUrl = String( sheet .getRange(targetRow, 8) .getValue() || "" ); if (photoUrl) { const match = photoUrl.match( /[?&]id=([^&]+)/ ); if (match) { try { DriveApp .getFileById(match[1]) .setTrashed(true); } catch (err) { // Jika foto tidak ditemukan, // komentar tetap dihapus. } } } /* * Hapus baris komentar */ sheet.deleteRow( targetRow ); return postResult_({ ok: true, summary: buildSummary_( sheet, 8, requesterEmail ) }); } /* ========================= UNKNOWN ACTION ========================= */ return postResult_({ ok: false, error: "Aksi tidak dikenali" }); }
